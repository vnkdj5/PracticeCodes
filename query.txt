select * FROM(
SELECT
  distinct note_typ.note_typ as note_typ,
  CASE
    WHEN lookup_typ_culture.lookup_desc IS NULL THEN LTRIM(RTRIM(note_typ.note_typ_desc))
    ELSE LTRIM(RTRIM(lookup_typ_culture.lookup_desc))
  END as note_typ_desc,
  Hospital.HospitalSer as Hospital
FROM inst_note_typ
  JOIN note_typ
    ON note_typ.note_typ = inst_note_typ.note_typ AND ( note_typ.note_typ < 800 or note_typ.note_typ > 1000)
  LEFT OUTER JOIN lookup_typ_culture
    ON lookup_typ_culture.table_name = 'note_typ' AND lookup_typ_culture.lookup_typ = note_typ.note_typ AND lookup_typ_culture.culture_cd = 'ENU'
  INNER JOIN inst
    ON inst.inst_id = inst_note_typ.inst_id AND inst.active_entry_ind = 'Y'
  LEFT OUTER JOIN Hospital on Hospital.HospitalSer = inst.hospital_ser
WHERE inst_note_typ.active_ind = 'Y'
) tmp
